Language Reference

      

        

          Capabilities are listed below. Use the Run button on each example to test it live.
        

        

          
          

          

        


        

          

            
print

            
Print builtin — outputs values to the result pane

            

              
print("hello") → hello

              
print(1 + 2) → 3

              
print(true) → true

            

          

          

            
arithmetic

            
Arithmetic operators — addition, subtraction, multiplication, and division

            

              
2 + 3 → 5

              
3 * 4 → 12

              
10 - 3 → 7

              
8 / 2 → 4

              
2 + 3 * 4 → 14

            

          

          

            
boolean

            
Boolean literals and logical operators

            

              
true → true

              
false → false

              
true and false → false

              
true or false → true

              
not true → false

              
not false → true

              
print(true and false) → false

              
print(true or false) → true

            

          

          

            
comparison

            
Comparison operators

            

              
1 == 1 → true

              
1 != 2 → true

              
1 < 2 → true

              
2 > 1 → true

              
1 <= 1 → true

              
1 >= 2 → false

            

          

          

            
control

            
Control flow constructs - let bindings, conditionals, and loops

            

              
let x = 1 in x → 1

              
if true then 1 else 2 end → 1

              
if false then 1 else 2 end → 2

              
let x = if true then 1 else 2 end in x → 1

              
if true then if false then 1 else 2 end else 3 end → 2

              
print(if true then 1 else 2 end) → 1

              
while false do print(1) end → 

              
while false do let x = 1 in print(x) end → 

            

          

          

            
function

            
Function definitions — named functions with fn and anonymous function expressions

            

              
fn add(a, b) = a + b
add(2, 3) → 5

              
fn add(a, b) = a + b
add(if true then 1 else 2 end, 3) → 4

              
(fn(a, b) = a + b)(2, 3) → 5

              
fn double(x) = x * 2
head(listMap(double, cons(1, cons(2, nil)))) → 2

              
let double = fn(x) = x * 2 in head(listMap(double, cons(1, cons(2, nil)))) → 2

              
head(listFilter(fn(x) = x > 1, cons(1, cons(2, cons(3, nil))))) → 2

              
listFold(fn(acc, x) = acc + x, 0, cons(1, cons(2, cons(3, nil)))) → 6

              
(fn() = 42)() → 42

            

          

          

            
list

            
List data structure — cons, head, tail, and nil

            

              
head(cons(1, nil)) → 1

              
tail(cons(1, nil)) → 

              
head(tail(cons(1, cons(2, nil)))) → 2

            

          

          

            
record

            
Record data structure — construction, field access, update, and presence checking

            

              
get(#{name: "x", value: 42}, "name") → x

              
get(#{name: "x", value: 42}, "value") → 42

              
has(#{name: "x"}, "name") → true

              
has(#{name: "x"}, "missing") → false

              
has(#{}, "anything") → false

            

          

          

            
list_ops

            
Higher-order list operations — map, filter, and fold

            

              
head(listMap(fn(x) = x + 1, cons(1, cons(2, nil)))) → 2

              
head(tail(listMap(fn(x) = x + 1, cons(1, cons(2, nil))))) → 3

              
length(listMap(fn(x) = x + 1, nil)) → 0

              
listFold(fn(acc, x) = acc + x, 0, cons(1, cons(2, cons(3, nil)))) → 6

              
listFold(fn(acc, x) = acc + x, 10, nil) → 10

              
head(listFilter(fn(x) = x > 1, cons(1, cons(2, cons(3, nil))))) → 2

              
length(listFilter(fn(x) = x > 1, cons(1, cons(2, cons(3, nil))))) → 2

              
length(listFilter(fn(x) = x > 10, cons(1, cons(2, nil)))) → 0

            

          

          

            
string

            
String operations — concatenation, length, and comparison

            

              
"hello" ++ " world" → hello world

              
length("abc") → 3

              
length("") → 0

              
length(nil) → 0

              
length(cons(1, cons(2, nil))) → 2

              
"abc" == "abc" → true

              
"abc" != "def" → true

              
"abc" == "def" → false

              
print("hello" ++ " " ++ "world") → hello world

            

          

          

            
letrec

            
Recursive local function bindings — letrec makes the binding visible in its own value expression

            

              
letrec factorial = fn(n) = if n <= 1 then 1 else n * factorial(n - 1) end in factorial(5) → 120

              
letrec fib = fn(n) = if n <= 1 then n else fib(n - 1) + fib(n - 2) end in fib(10) → 55

              
letrec count = fn(n) = if n <= 0 then 0 else 1 + count(n - 1) end in count(5) → 5

              
letrec double = fn(x) = x * 2 in double(3) → 6

              
letrec outer = fn(x) = letrec inner = fn(y) = x + y in inner(10) in outer(5) → 15
